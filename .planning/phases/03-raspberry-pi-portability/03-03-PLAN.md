  <name>Task 1: Verify built binaries are ARM aarch64 ELF</name>
  <files></files>
  <read_first>
    - .planning/phases/03-raspberry-pi-portability/03-RESEARCH.md (file command examples showing expected ARM aarch64 output)
    - docs/phase-03-docker-commands.md (Docker command to reuse for verification step)
  </read_first>
  <action>
    Run the file command on each built binary inside the linux/arm64 container to confirm they are ARM aarch64 ELF.
    Execute file on these paths relative to /work/telldus-core:
    - build/headless/service/telldusd
    - build/headless/client/libtelldus-core.so.2.1.3 (or the versioned .so file that exists)
    - build/headless/tdtool/tdtool
    Capture the full file command output for each binary.
    Assert each output contains "ARM aarch64" or "aarch64".
    If any binary is not ARM aarch64 (e.g., x86-64), flag it as a critical architecture mismatch.
  </action>
  <acceptance_criteria>
    - file output for telldusd contains "ARM aarch64"
    - file output for libtelldus-core.so contains "ARM aarch64"
    - file output for tdtool contains "ARM aarch64"
    - All three outputs are captured in the task log
  </acceptance_criteria>
  <verify>
    <automated>
      docker run --rm --platform linux/arm64 \
        -v "$(pwd)":/src:ro \
        -e DEBIAN_FRONTEND=noninteractive \
        debian:bookworm-slim \
        bash -c '
          set -e
          apt-get update -qq
          apt-get install -y -q cmake build-essential pkg-config libftdi1-dev libconfuse-dev libusb-1.0-0-dev libcppunit-dev file
          cp -r /src /work
          cd /work/telldus-core
          cmake -B build/headless -DFORCE_COMPILE_FROM_TRUNK=TRUE -DBUILD_TDTOOL=TRUE -DBUILD_TDADMIN=FALSE -DENABLE_TESTING=TRUE -DFTDI_ENGINE=libftdi
          cmake --build build/headless --parallel $(nproc)
          for binary in build/headless/service/telldusd build/headless/client/libtelldus-core.so.2.1.3 build/headless/tdtool/tdtool; do
            if [ -f "$binary" ]; then
              out=$(file "$binary")
              echo "$binary: $out"
              echo "$out" | grep -q "aarch64" || { echo "FAIL: $binary is not aarch64"; exit 1; }
            else
              echo "WARN: $binary not found, trying alternative"
              alt=$(ls build/headless/client/libtelldus-core.so* 2>/dev/null | head -n1)
              if [ -n "$alt" ]; then
                out=$(file "$alt")
                echo "$alt: $out"
                echo "$out" | grep -q "aarch64" || { echo "FAIL: $alt is not aarch64"; exit 1; }
              fi
            fi
          done
          echo "ARCH_OK"
        '
    </automated>
  </verify>
  <done>All three built binaries are confirmed as ARM aarch64 ELF format via the file command inside the container.</done>
